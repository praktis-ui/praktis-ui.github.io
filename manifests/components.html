// Живое чтение --btn-* — используется className из Button.module.css, чтобы гарантированно
// брать те же самые CSS-модульные объявления, которые видит настоящий Button.
function LiveSwatch({ label, varName, kind = 'fill' }: { label: string; varName: string; kind?: 'fill' | 'text' }) {
  const ref = useRef<HTMLDivElement>(null);
  const [value, setValue] = useState('');

  // Без dependency array — перечитывает при каждом ре-рендере, иначе после переключения
  // темы тулбаром Storybook значение останется от первого рендера (см. тот же паттерн
  // в ShahmatkaTokens.stories.tsx).
  useLayoutEffect(() => {
    if (ref.current) {
      setValue(getComputedStyle(ref.current).getPropertyValue(varName).trim());
    }
  });

  return (
    <div ref={ref} className={buttonStyles.button} style={{ display: 'flex', alignItems: 'center', gap: 12, background: 'none', border: 'none', padding: 0, height: 'auto' }}>
      <Chip value={`var(${varName})`} kind={kind} />
      <div style={{ display: 'flex', flexDirection: 'column', gap: 2, fontFamily: 'Inter, sans-serif' }}>
        <span style={{ fontSize: 13 }}>{label}</span>
        <code style={{ fontSize: 12, opacity: 0.6 }}>{value || '…'}</code>
      </div>
    </div>
  );
}

}

function IconGallery() {
  const [copied, setCopied] = useState<string | null>(null);

  const handleCopy = (name: string) => {
    navigator.clipboard.writeText(name).catch(() => {});
    setCopied(name);
    window.setTimeout(() => setCopied((current) => (current === name ? null : current)), 1200);
  };

  return (
    <div style={{ padding: 24 }}>
      <div
        style={{
          display: 'grid',
          gridTemplateColumns: 'repeat(auto-fill, minmax(120px, 1fr))',
          gap: 16,
        }}
      >
        {iconManifest.map(({ name, label, Icon }) => (
          <button
            key={name}
            type="button"
            onClick={() => handleCopy(name)}
            title={`Скопировать «${name}»`}
            style={{
              display: 'flex',
              flexDirection: 'column',
              alignItems: 'center',
              gap: 8,
              padding: 12,
              border: '1px solid rgba(128, 128, 128, 0.2)',
              borderRadius: 8,
              background: copied === name ? 'rgba(60, 131, 246, 0.1)' : 'transparent',
              cursor: 'pointer',
              font: 'inherit',
              color: 'inherit',
            }}
          >
            <Icon size={24} />
            <span

                fontSize: 11,
                color: copied === name ? '#3c83f6' : '#888',
                textAlign: 'center',
                wordBreak: 'break-word',
              }}
